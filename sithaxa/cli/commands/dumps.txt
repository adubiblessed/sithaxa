import typer
import os
from pathlib import Path
from dotenv import load_dotenv
from google import genai
from google.genai import types

load_dotenv()

app = typer.Typer(help="Explain code snippets using AI")


def get_api_key() -> str:
    """Get and validate API key."""
    api_key = os.environ.get("GEMINI_API_KEY")
    if not api_key:
        typer.echo("Error: GEMINI_API_KEY not found in environment variables.", err=True)
        typer.echo("Please set it in your .env file or environment.", err=True)
        raise typer.Exit(code=1)
    return api_key


@app.command()
def explain(
    code: str = typer.Option(None, "--code", "-c", help="Code snippet to explain"),
    file: Path = typer.Option(None, "--file", "-f", help="Path to file containing code"),
    verbose: bool = typer.Option(False, "--verbose", "-v", help="Show detailed output"),
):
    """
    Explain the provided code snippet.
    
    Examples:
        sithaxa explain -c "print('hello')"
        sithaxa explain -f script.py
        echo "def foo(): pass" | sithaxa explain
    """
    
    # Get code from different sources
    if file:
        if not file.exists():
            typer.echo(f"Error: File '{file}' not found.", err=True)
            raise typer.Exit(code=1)
        code_to_explain = file.read_text(encoding="utf-8")
        if verbose:
            typer.echo(f"Reading code from: {file}")
    elif code:
        code_to_explain = code
    else:
        # Read from stdin (for piping)
        import sys
        if not sys.stdin.isatty():
            code_to_explain = sys.stdin.read().strip()
        else:
            typer.echo("Error: Please provide code via --code, --file, or stdin.", err=True)
            raise typer.Exit(code=1)
    
    if not code_to_explain.strip():
        typer.echo("Error: No code provided.", err=True)
        raise typer.Exit(code=1)
    
    if verbose:
        typer.echo(f"\nCode to explain:\n{code_to_explain}\n")
        typer.echo("Generating explanation...\n")
    
    try:
        # Get API key
        api_key = get_api_key()
        client = genai.Client(api_key=api_key)
        
        # Create prompt
        prompt = f"""Explain the following code snippet in detail. Include:
- What the code does
- Key concepts used
- Any potential issues or improvements

Code:
```
{code_to_explain}
```

Explanation:"""
        
        # Generate response (simplified)
        response = client.models.generate_content(
            model="gemini-2.0-flash-exp",
            contents=prompt,
        )
        
        typer.echo("\n" + "="*50)
        typer.echo("EXPLANATION")
        typer.echo("="*50 + "\n")
        typer.echo(response.text)
        
    except Exception as e:
        typer.echo(f"Error: {e}", err=True)
        if verbose:
            import traceback
            traceback.print_exc()
        raise typer.Exit(code=1)


if __name__ == "__main__":
    app()

import typer
from sithaxa.cli.commands import (
    hello,
    commit,
    debug,
    explain,
    generate,
    learn,
    readme,
    recall,
)

# Main app
app = typer.Typer(
    help="Sithaxa - AI-powered local CLI coding assistant",
    add_completion=True,
)

# Add subcommands
app.add_typer(hello.app, name="hello", help="Greeting commands")
app.add_typer(commit.app, name="commit", help="Generate commit messages")
app.add_typer(debug.app, name="debug", help="Debug code issues")
app.add_typer(explain.app, name="explain", help="Explain code snippets")
app.add_typer(generate.app, name="generate", help="Generate code")
app.add_typer(learn.app, name="learn", help="Learn new concepts")
app.add_typer(readme.app, name="readme", help="Generate README files")
app.add_typer(recall.app, name="recall", help="Recall previous sessions")


@app.command()
def version():
    """Show Sithaxa version."""
    typer.echo("Sithaxa v1.0.0")


def main():
    """Entry point for the CLI application."""
    try:
        app()
    except KeyboardInterrupt:
        typer.echo("\n\nOperation cancelled by user.", err=True)
        raise typer.Exit(code=130)
    except Exception as e:
        typer.echo(f"\nError: {e}", err=True)
        raise typer.Exit(code=1)


if __name__ == "__main__":
    main()
